Actors block with actors, capabilities, purposes and grants

package nz.example.support

class Ticket {
    boolean internal
    int refundCents
}

actors Support {
    actor Customer
    actor Agent extends Customer
    agent AutoAgent extends Agent
    actor Manager

    capability ReadTicket on Ticket
    capability RaiseRefund on Ticket

    purpose audit
    purpose notifyCustomer

    grant Customer {
        permit ReadTicket when (!internal)
    }

    grant Agent {
        permit ReadTicket
        permit UpdateTicket when (refundCents > 0) obligation audit
        permit RaiseRefund when (refundCents > 0 && refundCents <= 5000) obligation audit obligation notifyCustomer
        forbid ResolveTicket
    }

    never_both { ReadTicket, ResolveTicket }
}

package nz.example.support

actors Support {
    actor Manager

    grant Manager {
        cedar {
            @why("internal tickets close at the manager's desk")
            forbid(principal is Support::Manager, action == "resolve") when { resource.internal };
        }
    }
}
