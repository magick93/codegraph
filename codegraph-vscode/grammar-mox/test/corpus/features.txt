Operation with parameters and body

package nz.example.org

class Team {
    op boolean any_member_earns_over(int threshold) {
        expr { members.any(m => m.salary > threshold) }
    }
    op int payroll() {
        rust { self.members.iter().map(|m| m.salary).sum() }
    }
    op void flush()
}
